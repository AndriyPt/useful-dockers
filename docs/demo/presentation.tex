\begin{frame}
    \frametitle{Рівняння Пеннеса}
    \begin{equation}
        \label{eqn:theory_thermo_pennes_eqn}
        \nabla \cdot \lambda_\theta \nabla \theta(x) + \omega \rho_b C_b (\theta_c - \theta(x)) + q_m  = 0
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Коефіцієнт теплопровідності}
    \begin{equation}
        \label{eqn:thermo_thermal_conductivity}
        \lambda_\theta(x) = \lambda_{\theta H}(x)\chi_H + \lambda_{\theta c}, x \in \Omega
    \end{equation}

    \begin{equation*}
        \lambda_{\theta H}(x) = (0,495 - \lambda_{\theta c}) \cdot
        cos\left[\frac{\pi}{2}\frac{||x-x_H||^2}{r_H^2}\right]
    \end{equation*}

    \begin{equation*}
        \lambda_{\theta c} = 0,19
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{Адитивне розщеплення оператора}
    \begin{equation}
        \label{eqn:thermo_pennes_separation_eqn}
        \lambda_{\theta c} \nabla^2 \theta(x) + \omega \rho_b C_b (\theta_c - \theta(x)) =
        - \nabla \lambda_{\theta H} (x) \cdot \nabla \theta(x) \cdot \chi_H
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{МГЕ -- інтегральне подання розв'язку}
    \begin{equation}
        \label{eqn:thermo_bem_direct_pennes}
        \begin{split}
            c(\xi)\theta(\xi) + \int \limits_{\Gamma}\theta(x)\hat{q}(x,\xi)d\Gamma(x) + \\
            + \int \limits_{H} \nabla{\lambda_{\theta H}(x)}\nabla{\theta(x)} \hat{\theta}(x,\xi) d H(x)
            = \int \limits_{\Gamma}q(x)\hat{\theta}(x,\xi)d\Gamma(x)
        \end{split}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{МГЕ -- зведення за формулою Гріна}
    \begin{equation}
        \label{eqn:thermo_bem_direct_pennes_simplified}
        \begin{split}
            c(\xi)\theta(\xi) + \int \limits_{\Gamma}\theta(x)\hat{q}(x,\xi)d\Gamma(x) = \\
            = \int \limits_{H}\theta(x) [ \nabla^2 \lambda_{\theta H}(x) \hat{\theta}(x,\xi) +
                \nabla \lambda_{\theta H}(x) \nabla \hat{\theta}(x,\xi) ] dH(x) + \\
            + \int \limits_{\Gamma}q(x)\hat{\theta}(x,\xi)d\Gamma(x)
        \end{split}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Рівняння рівноваги у переміщеннях}
    \begin{equation}
        \label{eqn:theory_elasticity_displacement_eqn}
        \mu u_{i,jj}(x) + (\lambda + \mu)u_{j,ji} + b_i(x) = 0
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Коефіцієнти Ляме}
    \begin{equation}
        \label{eqn:elasticity_lyame_lambda}
        \lambda(x) = \lambda_H(x)\chi_H + \lambda_c, x \in \Omega
    \end{equation}

    \begin{equation*}
        \lambda_H(x) = (3714,3 - \lambda_c) \cdot cos\left[\frac{\pi}{2}\frac{||x-x_H||^2}{r_H^2}\right]
    \end{equation*}

    \begin{equation*}
        \lambda_c = 660,1
    \end{equation*}

    \begin{equation}
        \label{eqn:elasticity_lyame_mu}
        \mu(x) = \mu_H(x)\chi_H + \mu_c, x \in \Omega
    \end{equation}

    \begin{equation*}
        \mu_H(x) = (3428,6 - \mu_c)cos\left[\frac{\pi}{2}\frac{||x-x_H||^2}{r_H^2}\right]
    \end{equation*}

    \begin{equation*}
        \mu_c = 404,6
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{Адитивне розщеплення оператора}
    \begin{equation}
        \label{eqn:elasticity_displacement_separation_eqn}
        \mu_c u_{i,jj}(x) + (\lambda_c + \mu_c)u_{j,ji} = -f_{ij,j}(\varepsilon(x))
    \end{equation}

    \begin{equation*}
        f_{ij,j}(\varepsilon(x)) = [ 2\mu_H(x)\varepsilon_{ij}(x) + \lambda_H(x) \delta_{ij} \varepsilon_{kk}(x) ]
        \chi_H
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{МПГЕ -- інтегральне подання розв'язку}
    \begin{equation*}
        \begin{split}
            u_i(\xi) = \int \limits_G \hat{u}_{ij}(x,\xi) g_j(x)d G(x)
            + \int \limits_H \hat{u}_{ij}(x,\xi)f_{jk,k}(\varepsilon(x))d H(x)
        \end{split}
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{МПГЕ -- зведення за формулою Гріна}
    \begin{equation}
        \label{eqn:elasticity_cobem_direct_elasticity_simplified}
        \begin{split}
            u_i(\xi) = \int \limits_G \hat{u}_{ij}(x,\xi) g_j(x)d G(x)
            - \int \limits_H \hat{u}_{ij,k}(x,\xi)f_{jk}(\varepsilon(x))d H(x)
        \end{split}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Система рівняннь}
    \begin{equation*}
        \label{eqn:theory_thermo_pennes_eqn}
        \nabla \cdot \lambda_\theta \nabla \theta(x) + \omega \rho_b C_b (\theta_c - \theta(x)) + q_m  = 0
    \end{equation*}

    \begin{equation}
        \label{eqn:theory_thermoelasticity_displacement_eqn}
        \mu u_{i,jj}(x) + (\lambda + \mu)u_{j,ji} - \beta \theta_{,i} = 0
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Адитивне розщеплення оператора}
    \begin{equation*}
        \mu_c u_{i,jj}(x) + (\lambda_c + \mu_c)u_{j,ji} = -f_{ij,j}(\varepsilon(x)) + \beta_{,j}(x) \theta(x) +
        \beta(x) \theta_{,j}(x)
    \end{equation*}
    \begin{equation*}
        f_{ij,j}(\varepsilon(x)) = [ 2\mu_H(x)\varepsilon_{ij}(x) + \lambda_H(x) \delta_{ij} \varepsilon_{kk}(x) ]
        \chi_H
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{МПГЕ -- інтегральне подання розв'язку}
    \begin{equation*}
        \begin{split}
            u_i(\xi) = \int \limits_G \hat{u}_{ij}(x,\xi) g_j(x)d G(x)
            + \int \limits_H \hat{u}_{ij}(x,\xi)f_{jk,k}(\varepsilon(x))d H(x) - \\
            - \int \limits_{\Omega} \hat{u}_{ij}(x, \xi) [\beta_{,j}(x) \theta(x) + \beta(x) \theta_{,j}(x)]d\Omega(x)
        \end{split}
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{МПГЕ -- зведення за формулою Гріна}
    \begin{equation*}
        \begin{split}
            u_i(\xi) = \int \limits_G \hat{u}_{ij}(x,\xi) g_j(x)d G(x)
            - \int \limits_H \hat{u}_{ij,k}(x,\xi)f_{jk}(\varepsilon(x))d H(x) - \\
            - \int \limits_{\Gamma} \hat{u}_{ij} \beta(x) \theta (x) n_j d \Gamma(x)
            + \int \limits_{\Omega} \theta(x)
            [\hat{u}_{ij,j}(x,\xi)\beta(x)-\hat{u}_{ij}(x,\xi)\beta_{,j}(x)]d\Omega(x)
        \end{split}
    \end{equation*}
\end{frame}
